1. DIAGRAMA DE CLASSES (MERMAID)
--------------------------------------------------------------------------------

classDiagram
    %% --- CONTROLADOR PRINCIPAL ---
    class CtrlPresentacio {
        - CtrlDomini ctrlDomini
        - VistaPrincipal vistaPrincipal
        - String currentUsername
        + inicializarPresentacio()
        + login(user, pass)
        + crearEnquesta(...)
        + registrarUsuari(...)
        + getEnquestesUsuari()
        + ... (mètodes delegats al domini)
    }

    %% --- VISTA PRINCIPAL (CONTENIDOR) ---
    class VistaPrincipal {
        - CtrlPresentacio iCtrlPresentacio
        - JFrame frameVista
        - JPanel panelContenidos
        - CardLayout cardLayout
        - VistaLogin vistaLogin
        - VistaRegistro vistaRegistro
        - VistaMenuPrincipal vistaMenuPrincipal
        - VistaGestionEnquestes vistaGestionEnquestes
        - VistaGestionarRespostes vistaGestionarRespostes
        - VistaAnalisi vistaAnalisi
        + hacerVisible()
        + mostrarVista(String nombreVista)
    }

    %% --- GESTOR D'ESDEVENIMENTS ---
    class MyActionListener {
        <<ActionListener>>
        - CtrlPresentacio ctrlPresentacio
        - VistaPrincipal vistaPrincipal
        - Object context
        + actionPerformed(ActionEvent e)
        - handleAction(Action action, ActionEvent e)
        - handleLogin()
        - handleCrearPregunta()
        - handleModificarRespostaIndividual()
        - ... (handlers específics)
    }

    %% --- VISTES (PANELS) ---
    class VistaLogin {
        - JTextField userText
        - JPasswordField passText
        + getUsername()
        + getPassword()
    }
    class VistaRegistro {
        - JTextField userText
        - JPasswordField passText
        + getUsername()
    }
    class VistaMenuPrincipal {
        + inicializar()
    }
    class VistaGestionEnquestes {
        + JList listEnquestes
        + actualizarLista()
    }
    class VistaGestionarRespostes {
        + JList listEnquestes
        + actualizarLista()
    }
    class VistaAnalisi {
        + inicializar()
    }

    %% --- DIÀLEGS (POPUPS) ---
    class DialogoGestionPreguntes {
        - String idEnquesta
        + cargarPreguntes()
        + getSelectedPreguntaId()
    }
    class DialogoGestionarRespostes {
        - String idEnquesta
        + actualizarVista()
        + getResposta(id)
    }
    class DialogoCrearPregunta {
        + getPreguntaText()
        + getTipus()
    }
    class DialogoParticipants
    class DialogoResponderEnquesta
    class DialogoSeleccionarEnquesta

    %% --- RELACIONS ---
    
    CtrlPresentacio --> VistaPrincipal : crea i gestiona
    
    VistaPrincipal *-- VistaLogin
    VistaPrincipal *-- VistaRegistro
    VistaPrincipal *-- VistaMenuPrincipal
    VistaPrincipal *-- VistaGestionEnquestes
    VistaPrincipal *-- VistaGestionarRespostes
    VistaPrincipal *-- VistaAnalisi

    VistaLogin --> CtrlPresentacio
    VistaRegistro --> CtrlPresentacio
    VistaMenuPrincipal --> CtrlPresentacio
    VistaGestionEnquestes --> CtrlPresentacio
    VistaGestionarRespostes --> CtrlPresentacio

    VistaLogin ..> MyActionListener : usa
    VistaRegistro ..> MyActionListener : usa
    VistaMenuPrincipal ..> MyActionListener : usa
    VistaGestionEnquestes ..> MyActionListener : usa
    DialogoGestionPreguntes ..> MyActionListener : usa
    DialogoGestionarRespostes ..> MyActionListener : usa

    MyActionListener --> CtrlPresentacio : crida lògica
    MyActionListener --> VistaPrincipal : navega
    MyActionListener ..> DialogoGestionPreguntes : manipula (context)
    MyActionListener ..> DialogoGestionarRespostes : manipula (context)

    MyActionListener ..> DialogoGestionPreguntes : crea
    MyActionListener ..> DialogoGestionarRespostes : crea
    MyActionListener ..> DialogoCrearPregunta : crea


2. JERARQUIA ESTRUCTURAL (QUI CONTÉ A QUI)
--------------------------------------------------------------------------------

[ CtrlPresentacio ]
       │
       │ (Crea e Inicialitza)
       ▼
[ VistaPrincipal ]
       │
       │ (Conté i Gestiona amb CardLayout)
       │
       ├───> [ VistaLogin ]
       ├───> [ VistaRegistro ]
       ├───> [ VistaMenuPrincipal ]
       ├───> [ VistaGestionEnquestes ]
       ├───> [ VistaGestionarRespostes ]
       └───> [ VistaAnalisi ]


3. FLUX D'INTERACCIÓ (QUÈ PASSA EN FER CLIC)
--------------------------------------------------------------------------------

      USUARI
         │
         ▼ (Fa Clic en un Botó)
         │
[ Vista o Diàleg ] 
         │
         │ (Envia Esdeveniment 'ActionCommand')
         │
         ▼
[ MyActionListener ] ──────────────────────────────────────────────┐
         │                                                         │
         │ (Decideix què fer segons el comando)                    │
         │                                                         │
         ├─── SI ÉS NAVEGACIÓ (Ex: "Anar a Menú")                  │
         │      │                                                  │
         │      ▼                                                  │
         │   [ VistaPrincipal ]                                    │
         │      └──> .mostrarVista("MENU")                         │
         │                                                         │
         ├─── SI ÉS LÒGICA (Ex: "Esborrar Pregunta")               │
         │      │                                                  │
         │      ▼                                                  │
         │   [ CtrlPresentacio ] ──────────────────────────> [ DOMINI ]
         │      └──> .eliminarPregunta(...)                        │
         │                                                         │
         ├─── SI ÉS GESTIÓ UI (Ex: "Refrescar Llista")             │
         │      │                                                  │
         │      ▼                                                  │
         │   [ Context (Vista/Diàleg) ]                            │
         │      └──> .actualizarLista() / .cargarPreguntes()       │
         │                                                         │
         └─── SI ÉS OBRIR FINESTRA (Ex: "Gestionar Preguntes")     │
                │                                                  │
                ▼                                                  │
             [ Crea Nou Diàleg ] <─────────────────────────────────┘
                (DialogoGestionPreguntes, etc.)

--------------------------------------------------------------------------------

       │ (Botó "Gestionar Preguntes")
       ▼
[ MyActionListener ] ──(Crea)──> [ DialogoGestionPreguntes ]

                                            ▼
                                   [ MyActionListener ]
                                            │
                                            │ (Llegeix selecció)
                                            ▼
                                   [ DialogoGestionPreguntes ]
                                            │
                                            │ (Executa acció)
                                            ▼
                                   [ CtrlPresentacio ]
